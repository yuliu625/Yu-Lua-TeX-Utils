% Academic Standard
\usepackage{booktabs}  % \toprule \midrule \bottomrule
% Adds vertical padding and different line weights.

\usepackage{tabularx}  % \begin{tabularx}{\textwidth}{X}

\usepackage{xltabular}  % The modern replacement for ltablex.

\usepackage{makecell}   % \makecell{line1 \\ line2} and \thead

\usepackage{siunitx}  % S column for decimal alignment: \begin{tabular}{S}
