% yu for figures

\NewDocumentCommand{\yuQuickInsertFigure}{ O{1} m o o }{
    % #1: size
    % #2: path
    % #3: caption
    % #4: label
    \begin{figure}[htbp]
        \centering
        \includegraphics[width=#1\textwidth]{#2}
        \IfValueT{#3}{
            \caption{#3}
        }
        \IfValueTF{#4}{
            \label{#4}}{\label{fig:#2}
        }
    \end{figure}
}
